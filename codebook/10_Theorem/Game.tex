\begin{itemize}
\item Nim variants (rules $\Rightarrow$ the player to move loses iff)
\begin{itemize}
  \item Staircase Nim (coins on steps $1, \ldots, k$; move any number of coins from step $i$ to step $i - 1$; step $0$ is out of the game) $\Rightarrow$ the xor of the odd steps is $0$.
  \item Moore's Nim$_k$ (take from up to $k$ piles at once) $\Rightarrow$ for every bit, the number of piles having that bit is divisible by $k + 1$.
  \item Wythoff (two piles $a \le b$; take from one pile, or the same amount from both) $\Rightarrow$ $a = \lfloor k\varphi \rfloor$, $b = a + k$ for some $k \ge 0$, $\varphi = \frac{1+\sqrt5}{2}$.
\end{itemize}

\item Misère games (whoever makes the last move loses; rules $\Rightarrow$ the player to move loses iff)
\begin{itemize}
  \item Misère Nim $\Rightarrow$ (some pile $> 1$ and $\bigoplus a_i = 0$) or (all piles $\le 1$ and $\bigoplus a_i \ne 0$).
  \item SJ theorem (a sum of games in which every position with $SG = 0$ is terminal) $\Rightarrow$ (some $SG_i > 1$ and $\bigoplus SG_i = 0$) or (every $SG_i \le 1$ and $\bigoplus SG_i \ne 0$).
\end{itemize}

\item Undirected vertex geography (a token starts on $v$; move it to an unvisited neighbour; whoever cannot move loses) $\Rightarrow$ the player to move loses iff some maximum matching misses $v$, i.e.\ $\nu(G - v) = \nu(G)$. Otherwise move along the edge at $v$ of any maximum matching of $G$.

\item Green Hackenbush (cut an edge; whatever is no longer connected to the ground disappears) $\Rightarrow$ a tree whose root $r$ stands on the ground has $SG = g(r)$, where $g(v) = \bigoplus_{c \text{ child of } v} (g(c) + 1)$ ($g = 0$ at leaves); several trees: xor.
\end{itemize}
