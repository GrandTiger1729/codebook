\begin{itemize}
\item Ordinary Generating Function
$A(x) = \sum_{i\ge 0} a_ix^i$
\begin{itemize}
    \item $xA'(x)            \Rightarrow na_n$
    \item $\frac{A(x)}{1-x}  \Rightarrow \sum_{i=0}^{n} a_i$
\end{itemize}
\item Exponential Generating Function
$A(x) = \sum_{i\ge 0} \frac{a_i}{i!}x^i$
\begin{itemize}
    \item $A^{(k)}(x)        \Rightarrow a_{n+k}$
    \item $xA(x)             \Rightarrow na_{n-1}$
    \item $A(x)B(x)          \Rightarrow \sum_{i=0}^{n} \binom{n}{i}a_ib_{n-i}$
\end{itemize}
\item Special Generating Function
\begin{itemize}
    \item $\frac{1}{(1-x)^n} = \sum_{i\ge 0} \binom{i+n-1}{n-1}x^i$
    \item $-\ln(1-x) = \sum_{i\ge 1} \frac{x^i}{i}$, \quad $\ln(1+x) = \sum_{i\ge 1} \frac{(-1)^{i+1}x^i}{i}$
    \item $\frac{d^k}{dx^k}\left(-\ln(1-x)\right) = \frac{(k-1)!}{(1-x)^k}$, \quad $(\ln A(x))' = \frac{A'(x)}{A(x)}$
    \item $\frac{(-\ln(1-x))^k}{k!} = \sum_{n} c(n, k)\frac{x^n}{n!}$ (Stirling first kind)
    \item Power sums $p_k = \sum_i a_i^k$: $\sum_{k\ge 1} p_k x^k = -x\frac{d}{dx}\ln\prod_i(1-a_ix)$
    \item $\sum_{m \mid n} a_nx^n = \frac{1}{m}\sum_{j=0}^{m-1} A(\omega^jx)$, $\omega$ a primitive $m$-th root of unity (mod $p$ needs $m \mid p - 1$)
\end{itemize}
\item Lagrange inversion

If $F(G(x)) = x$ with $F(0) = 0$, $F'(0) \ne 0$, then for $n \ge 1$: $[x^n]\, G(x)^k = \frac{k}{n} [x^{n-k}] \left(\frac{x}{F(x)}\right)^n$ and $[x^n]\, H(G(x)) = \frac 1 n [x^{n-1}]\, H'(x) \left(\frac{x}{F(x)}\right)^n$.
\item Falling factorials

$x^{\underline k} = x(x - 1) \cdots (x - k + 1)$.

From $f(0), \ldots, f(n)$, the coefficients of $f(x) = \sum_k c_k x^{\underline k}$ can be found by one convolution: $\sum_k c_k x^k = e^{-x} \sum_i f(i) \frac{x^i}{i!}$. Useful because $\sum_{x=0}^{m-1} x^{\underline k} = \frac{m^{\underline{k+1}}}{k+1}$.
\item Linear recurrences

\begin{itemize}
  \item $a_n = \sum_{i=1}^{d} c_i a_{n-i}$ ($n \ge d$) iff $\sum a_n x^n = \frac{P(x)}{Q(x)}$ with $Q = 1 - \sum c_i x^i$, $\deg P < d$.
  \item If $a$, $b$ have orders $d_a$, $d_b$: $a_n + b_n$ has order $\le d_a + d_b$, $a_n b_n$ order $\le d_a d_b$, $\sum_{i \le n} a_i$ order $\le d_a + 1$; Berlekamp–Massey on twice that many terms finds the recurrence.
\end{itemize}
\end{itemize}
