\begin{itemize}
\item Determinant
\begin{itemize}
  \item Vandermonde: $\det [x_i^{\,j}]_{0 \le i, j < n} = \prod_{i < j} (x_j - x_i)$.
  \item Schur complement: $\det \begin{bmatrix} A & B \\ C & D \end{bmatrix} = \det A \cdot \det(D - CA^{-1}B)$.
  \item Matrix determinant lemma: $\det(A + uv^\mathsf{T}) = (1 + v^\mathsf{T}A^{-1}u) \det A$. Sherman–Morrison: $(A + uv^\mathsf{T})^{-1} = A^{-1} - \frac{A^{-1}uv^\mathsf{T}A^{-1}}{1 + v^\mathsf{T}A^{-1}u}$.
  \item Hadamard: $|\det A| \le \prod_i \lVert \text{row}_i \rVert_2$; bounds how many primes CRT needs for an exact integer determinant.
  \item Modulo a composite $m$: eliminate with Euclid's algorithm on two rows (\texttt{DeterminantAnyMod}).
\end{itemize}

\item Powers and the characteristic polynomial $p(x) = \det(xI - A)$
\begin{itemize}
  \item Cayley–Hamilton: $p(A) = 0$, so $A^k = r(A)$ with $r = x^k \bmod p$; $u^\mathsf{T} A^k v$ is then a linear recurrence with characteristic polynomial $p$ (feed $2n$ terms to Berlekamp–Massey).
  \item $(A^k)_{ij}$ counts walks of length $k$ from $i$ to $j$; $\operatorname{tr} A^k = \sum \lambda_i^k$ counts closed walks. The power sums $\operatorname{tr} A^k$ and the coefficients of $p$ determine each other (Newton's identities, Polynomial chapter).
\end{itemize}

\item Over a finite field $\mathbb{F}_q$
\begin{itemize}
  \item $Ax = b$ ($n$ unknowns) has $0$ or $q^{\,n - \operatorname{rank} A}$ solutions.
  \item Invertible $n \times n$ matrices: $\prod_{i=0}^{n-1} (q^n - q^i)$. Subspaces of dimension $k$ in $\mathbb{F}_q^n$: $\binom{n}{k}_q = \prod_{i=0}^{k-1} \frac{q^{n-i} - 1}{q^{i+1} - 1}$.
  \item Xor basis: $n$ vectors of rank $r$ span $2^r$ values, each the xor of exactly $2^{n-r}$ subsets. Rank over $\mathbb{F}_2$ with bitsets: $O(n^3 / 64)$.
\end{itemize}

\item Randomized checks
\begin{itemize}
  \item Freivalds: test $AB = C$ by comparing $A(Bv)$ with $Cv$ for a random $v$, $O(n^2)$; a wrong $C$ passes with probability $\le 1/2$ ($v \in \{0, 1\}^n$) or $\le 1/|S|$ ($v \in S^n$ over a field).
  \item Schwartz–Zippel: a non-zero polynomial of total degree $d$ vanishes at a uniformly random point of $S^n$ with probability $\le d/|S|$ (e.g.\ $\det$ of a matrix of variables, Tutte's matrix).
\end{itemize}
\end{itemize}
