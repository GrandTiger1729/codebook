\begin{itemize}
\item Bernoulli numbers (convention $B_1 = +\frac{1}{2}$)

$B_0=1,B_1=\frac{1}{2},B_2=\frac{1}{6},B_3=0$

$\displaystyle\sum_{j=0}^m\binom{m+1}{j}B_j=m+1$, EGF is $\frac{x}{1 - e^{-x}}=\displaystyle\sum_{n=0}^\infty B_n\frac{x^n}{n!}$.

$S_m(n)=\displaystyle\sum_{k=1}^nk^m=\frac{1}{m+1}\sum_{k=0}^m\binom{m+1}{k}B_kn^{m+1-k}$

\item Stirling numbers of the first kind
Permutations of $n$ elements with exactly $k$ cycles.

$c(n, k) = c(n - 1, k - 1) + (n - 1)c(n - 1, k)$

$x(x + 1)\cdots(x + n - 1) = \sum_{k=0}^{n} c(n, k) x^k$

\item Stirling numbers of the second kind
Partitions of $n$ distinct elements into exactly $k$ groups.

$S(n, k) = S(n - 1, k - 1) + kS(n - 1, k), S(n, 1) = S(n, n) = 1$

$S(n, k) = \frac{1}{k!}\sum_{i=0}^{k}(-1)^{k-i}{k \choose i}i^n$

$x^n     = \sum_{i=0}^{n} S(n, i) (x)_i$

\item Bell numbers

$B_{n+1} = \sum_{k=0}^{n} \binom nk B_k$, $B_n = \sum_k S(n, k)$, EGF $\exp(e^x-1)$.

\item Pentagonal number theorem

$\displaystyle\prod_{n=1}^{\infty}(1-x^n)=1+\sum_{k=1}^{\infty}(-1)^k\left(x^{k(3k+1)/2} + x^{k(3k-1)/2}\right)$

\item Catalan numbers ($k = 2$; general $k$: Fuss--Catalan)

$C^{(k)}_n$: number of $k$-ary trees with $n$ internal nodes, where trees are compared as in the Cycle Lemma (children are ordered).

$C^{(k)}_n = \displaystyle \frac{1}{(k - 1)n + 1}\binom{kn}{n}$, \quad GF: $C^{(k)}(x) = 1 + x [C^{(k)}(x)]^k$

\item Narayana numbers

$N(n,k) = \frac1n \binom nk \binom n{k-1}$: Dyck paths of semilength $n$ with $k$ peaks (ordered trees with $n$ edges and $k$ leaves). Rows: \texttt{1; 1 1; 1 3 1; 1 6 6 1; 1 10 20 10 1}.

\item Eulerian numbers

Number of permutations $\pi \in S_n$ in which exactly $k$ elements are greater than the previous element. $k$ $j$:s s.t. $\pi(j)>\pi(j+1)$, $k+1$ $j$:s s.t. $\pi(j)\geq j$, $k$ $j$:s s.t. $\pi(j)>j$.

$E(n,k) = (n-k)E(n-1,k-1) + (k+1)E(n-1,k)$

$E(n,0) = E(n,n-1) = 1$

$E(n,k) = \sum_{j=0}^k(-1)^j\binom{n+1}{j}(k+1-j)^n$

\end{itemize}
