\begin{itemize}
\item Bernoulli numbers (convention $B_1 = +\frac{1}{2}$)

\begin{itemize}
  \item $B_0=1,B_1=\frac{1}{2},B_2=\frac{1}{6},B_3=0$
  \item $\displaystyle\sum_{j=0}^m\binom{m+1}{j}B_j=m+1$, EGF is $\displaystyle\frac{x}{1 - e^{-x}}=\displaystyle\sum_{n=0}^\infty B_n\frac{x^n}{n!}$.
  \item $S_m(n)=\displaystyle\sum_{k=1}^nk^m=\frac{1}{m+1}\sum_{k=0}^m\binom{m+1}{k}B_kn^{m+1-k}$
\end{itemize}

\item Stirling numbers of the first kind

$c(n, k) \, = $ number of permutations of $n$ elements with exactly $k$ cycles.

\begin{itemize}
  \item $c(n, k) = c(n - 1, k - 1) + (n - 1)c(n - 1, k)$
  \item $x(x + 1)\cdots(x + n - 1) = \sum_{k=0}^{n} c(n, k) x^k$
\end{itemize}

\item Stirling numbers of the second kind

$S(n, k) \, = $ number of partitions of $n$ distinct elements into exactly $k$ groups.

\begin{itemize}
  \item $S(n, k) = S(n - 1, k - 1) + kS(n - 1, k), S(n, 1) = S(n, n) = 1$
  \item $S(n, k) = \frac{1}{k!}\sum_{i=0}^{k}(-1)^{k-i}{k \choose i}i^n$
  \item $x^n     = \sum_{i=0}^{n} S(n, i) (x)_i$
\end{itemize}

\item Bell numbers

$B_{n+1} = \sum_{k=0}^{n} \binom nk B_k$, $B_n = \sum_k S(n, k)$, EGF $\exp(e^x-1)$.

\item Pentagonal number theorem

$\displaystyle\prod_{n=1}^{\infty}(1-x^n)=1+\sum_{k=1}^{\infty}(-1)^k\left(x^{k(3k+1)/2} + x^{k(3k-1)/2}\right)$

\item Catalan numbers ($k = 2$; general $k$: Fuss--Catalan)

$C^{(k)}_n$: number of $k$-ary trees with $n$ internal nodes, where trees are compared as in the Cycle Lemma (children are ordered).

\begin{itemize}
  \item $C^{(k)}_n = \frac{1}{(k - 1)n + 1}\binom{kn}{n}$, \quad GF: $C^{(k)}(x) = 1 + x [C^{(k)}(x)]^k$
\end{itemize}

\item Narayana numbers

$N(n, k) = \frac{1}{n} \binom{n}{k} \binom{n}{k-1}$ = number of balanced sequences of $n$ bracket pairs with exactly $k$ occurrences of ``()'' (equivalently, ordered trees with $n$ edges and $k$ leaves); $\sum_k N(n, k) = C_n$. Rows $n = 1..5$: 1; 1 1; 1 3 1; 1 6 6 1; 1 10 20 10 1.

\item Eulerian numbers

$E(n, k)$: permutations of $n$ with $k$ ascents ($\pi(j) < \pi(j+1)$); same count for $k$ descents, $k$ excedances ($\pi(j) > j$), $k+1$ weak excedances ($\pi(j) \ge j$).
\begin{itemize}
  \item $E(n,k) = (k+1)E(n-1,k) + (n-k)E(n-1,k-1)$
  \item $E(n,k) = \sum_{j=0}^{k} (-1)^j \binom{n+1}{j} (k+1-j)^n$, $E(n,0) = E(n,n-1) = 1$.
\end{itemize}

\end{itemize}
