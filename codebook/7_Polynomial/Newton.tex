Given $F(x) = \sum_{i=0}^{\infty} \alpha_i (x - \beta)^i$ for some constant $\beta$, a polynomial $P$ with $F(P) = 0$ can be found iteratively. If $Q_k$ satisfies $F(Q_k) \equiv 0 \pmod {x^{2^k}}$, then

$$ Q_{k+1} = Q_k - \frac{F(Q_k)}{F^\prime(Q_k)} \pmod {x^{2^{k+1}}} $$

\begin{itemize}
    \item $A^{-1}$: $B_{k+1} = B_k(2-AB_k) \pmod{x^{2^{k+1}}}$; $A_0 \ne 0$, $B_0 = A_0^{-1}$
    \item $\ln A$: $(\ln A)' = \frac{A'}{A}$; $A_0 = 1$
    \item $\exp A$: $B_{k+1} = B_k(1 + A - \ln B_k) \pmod{x^{2^{k+1}}}$; $A_0 = 0$, $B_0 = 1$
    \item $\sqrt A$: $B_{k+1} = \frac 1 2 (B_k + AB_k^{-1}) \pmod{x^{2^{k+1}}}$; $B_0 = \sqrt{A_0}$
\end{itemize}
