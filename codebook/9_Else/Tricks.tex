\begin{itemize}
\item Reformulate
\begin{itemize}
  \item Contribution: a sum over pairs / subarrays / subsets is a sum over elements of how often each counts. Sum of max over subarrays: previous greater with $\ge$, next greater with $>$ (monotonic stack). Tree: $\sum$ dist $= \sum_{e} sz(n - sz)$.
  \item Complement, or inclusion--exclusion over the violated constraints. Fix the max / min / middle element (sort, insert in that order).
  \item Reverse time: deletions become insertions (DSU). ``All to one'': reverse the edges.
  \item Manhattan $\to$ Chebyshev: $(x, y) \mapsto (x + y, x - y)$, then $|\Delta x| + |\Delta y| = \max(|\Delta u|, |\Delta v|)$.
  \item Exchange argument: sort by comparing two adjacent items in both orders; check it is a strict weak order.
  \item Lexicographically smallest: fix positions greedily, each time checking that the rest is still feasible.
  \item $E[X] = \sum_{k \ge 1} P(X \ge k)$ for $X \in \mathbb{N}$; linearity needs no independence.
\end{itemize}

\item Offline
\begin{itemize}
  \item Binary search the answer; for many queries, parallel binary search (each round replays the updates once).
  \item Segment tree over time + rollback DSU (union by size, no path compression): deletions in $O(n\log^2 n)$.
  \item CDQ: split by time, left updates answer right queries by sort + BIT; 3D dominance in $O(n \log^2 n)$.
  \item Sort queries by threshold and add elements in that order. Kruskal reconstruction tree: each union gets a new node of weight $w$; vertices reachable from $v$ with edges $\le w$ are a subtree (binary lift from $v$).
  \item Distinct values in $[l, r]$: sweep $r$, keep $+1$ only at each value's last occurrence.
\end{itemize}

\item Data structures
\begin{itemize}
  \item Small to large: always merge the smaller set into the larger, $O(n \log n)$ moves. On trees, keep the heavy child's set.
  \item DFS order: subtree $= [tin_v, tout_v]$. Point update on $v$ with root-to-$u$ path query: add on $v$'s subtree, read at $tin_u$.
  \item LCA by DFS order: for $tin_u < tin_v$, the parent of the shallowest vertex at positions $(tin_u, tin_v]$ (sparse table).
  \item HLD: heavy child first in the DFS, so a path is $O(\log n)$ ranges; jump from the chain whose head is deeper.
  \item BIT range add + range sum: $\sum_{i \le x} a_i = (x + 1)\sum_{i \le x} d_i - \sum_{i \le x} i\, d_i$ ($d$ = difference array).
  \item BIT $k$-th: for $j$ from high to low, if $pos + 2^j \le n$ and $t[pos + 2^j] < k$ then $pos \mathrel{+}= 2^j$, $k \mathrel{-}= t[pos]$; answer $pos + 1$.
  \item Persistent segment tree over prefixes: $k$-th in $a[l..r]$ by descending trees $r$ and $l - 1$ together.
  \item Sliding window min: monotone deque. Sqrt split: heavy $= \deg \ge \sqrt m$.
  \item XOR hashing: random 64-bit per value; a multiset is its sum, ``every count even'' is XOR $= 0$ (compare prefix XORs).
\end{itemize}

\item DP
\begin{itemize}
  \item Divide and conquer when $opt(i) \le opt(i + 1)$: $O(kn\log n)$. Knuth: $opt_{i,j-1} \le opt_{i,j} \le opt_{i+1,j}$: $O(n^2)$.
  \item Aliens (WQS): if the best with exactly $k$ items is convex in $k$, charge $\lambda$ per item, drop $k$, binary search $\lambda$; answer $= g(\lambda) + \lambda k$. Break ties towards the fewest (or most) items consistently.
  \item Slope trick: keep the breakpoints of a convex piecewise linear function in heaps. Min $\sum|b_i - a_i|$ with $b$ non-decreasing: push $a_i$ to a max-heap; if top $> a_i$: ans $\mathrel{+}=$ top $- a_i$, pop, push $a_i$.
  \item Sum over submasks (SOS): for each bit $i$, for each mask with bit $i$: $f[m] \mathrel{+}= f[m \oplus 2^i]$. Submasks: \texttt{for (s = m; s; s = (s - 1) \& m)}, $O(3^n)$ in total.
  \item Rerooting: DP down, then up using prefix / suffix merges of the children. Tree knapsack capped at $k$: merging sizes $a, b$ costs $\min(a,k)\min(b,k)$, $O(nk)$ total.
  \item Meet in the middle $2^{n/2}$; $(\min, +)$ matrix power for paths of exactly $k$ edges.
\end{itemize}

\item Graphs
\begin{itemize}
  \item 0-1 BFS with a deque. Layered graph (vertex, extra state) for ``$k$ free edges'', parity, \ldots
  \item Difference constraints $x_v - x_u \le w$: edge $u \to v$ of weight $w$, shortest paths from a virtual source; a negative cycle means infeasible. Johnson: $w'(u, v) = w + h_u - h_v \ge 0$.
  \item DSU with parity: online bipartiteness / ``same or different'' constraints.
  \item Functional graph: each component is one cycle with trees; $k$-th successor by binary lifting.
  \item Tree diameter: farthest vertex from any vertex, then farthest from that one; the centre is its middle.
  \item Second best MST: add a non-tree edge, remove the max edge on the tree path.
\end{itemize}

\item Numbers and bits
\begin{itemize}
  \item $inv_i = -\lfloor p / i \rfloor \cdot inv_{p \bmod i}$; inverse factorials backwards; many inverses by prefix products.
  \item $\sum_{i \le n} n / i \approx n \ln n$: loop over multiples. $\sum_{d \mid g} \varphi(d) = g$, $[g = 1] = \sum_{d \mid g} \mu(d)$.
  \item $a + b = (a \oplus b) + 2(a \mathbin{\&} b)$. Gray code $i \oplus (i \gg 1)$. Next mask with the same popcount: \texttt{c = x \& -x, r = x + c, x = (((r \^{} x) >> 2) / c) | r}.
\end{itemize}

\item Randomization
\begin{itemize}
  \item Shuffle the input (random incremental, e.g.\ enclosing circle, is $O(n)$ expected). Element occurring $> n/2$ times: try random elements. Hash collisions appear after about $\sqrt{\text{range}}$ values (birthday).
\end{itemize}
\end{itemize}
